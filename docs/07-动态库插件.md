# 07 — 动态库与插件系统

## 1. 核心概念

动态库（Windows `.dll` / Linux `.so` / macOS `.dylib`）是运行时加载的共享代码。插件系统基于动态库实现运行时扩展——主程序定义接口，插件以动态库形式实现接口，无需重新编译主程序即可扩展功能。

**为什么用动态库/插件？**
- 代码复用：多个程序共享同一个库
- 热插拔：不重启程序即可加载/卸载功能模块
- 模块化：编译隔离，减少构建时间
- 第三方扩展：允许用户编写自定义插件

## 2. 快速上手

### 2.1 构建动态库（CMake）

```cmake
cmake_minimum_required(VERSION 3.16)
project(MyPlugin)

add_library(my_plugin SHARED
    plugin.cpp
)

target_include_directories(my_plugin PUBLIC include)
```

### 2.2 编写插件（C接口导出）

```cpp
// plugin.h —— 插件接口（主程序和插件都include）
#pragma once

#ifdef _WIN32
    #define PLUGIN_API __declspec(dllexport)
#else
    #define PLUGIN_API __attribute__((visibility("default")))
#endif

extern "C" {
    PLUGIN_API const char* plugin_name();
    PLUGIN_API int plugin_version();
    PLUGIN_API int execute(int a, int b);
}
```

```cpp
// plugin.cpp —— 插件实现
#include "plugin.h"

extern "C" {

PLUGIN_API const char* plugin_name() {
    return "加法插件";
}

PLUGIN_API int plugin_version() {
    return 1;
}

PLUGIN_API int execute(int a, int b) {
    return a + b;
}

}
```

### 2.3 加载插件（主程序）

```cpp
// main.cpp —— 主程序加载插件
#ifdef _WIN32
    #include <windows.h>
#else
    #include <dlfcn.h>
#endif

#include <iostream>
#include <string>

// 函数指针类型
typedef const char* (*PluginNameFunc)();
typedef int (*PluginVersionFunc)();
typedef int (*ExecuteFunc)(int, int);

class DynamicLib {
public:
    explicit DynamicLib(const std::string& path) {
#ifdef _WIN32
        handle_ = LoadLibraryA(path.c_str());
#else
        handle_ = dlopen(path.c_str(), RTLD_LAZY);
#endif
        if (!handle_) {
            throw std::runtime_error("加载动态库失败: " + path);
        }
    }

    ~DynamicLib() {
        if (handle_) {
#ifdef _WIN32
            FreeLibrary((HMODULE)handle_);
#else
            dlclose(handle_);
#endif
        }
    }

    // 禁止拷贝
    DynamicLib(const DynamicLib&) = delete;
    DynamicLib& operator=(const DynamicLib&) = delete;

    // 允许移动
    DynamicLib(DynamicLib&& other) noexcept : handle_(other.handle_) {
        other.handle_ = nullptr;
    }

    template<typename FuncType>
    FuncType get_function(const std::string& name) {
#ifdef _WIN32
        return reinterpret_cast<FuncType>(GetProcAddress((HMODULE)handle_, name.c_str()));
#else
        return reinterpret_cast<FuncType>(dlsym(handle_, name.c_str()));
#endif
    }

private:
    void* handle_ = nullptr;
};

// 使用
int main() {
#ifdef _WIN32
    const std::string lib_path = "my_plugin.dll";
#else
    const std::string lib_path = "./libmy_plugin.so";
#endif

    try {
        DynamicLib lib(lib_path);

        auto name_fn    = lib.get_function<PluginNameFunc>("plugin_name");
        auto version_fn = lib.get_function<PluginVersionFunc>("plugin_version");
        auto exec_fn    = lib.get_function<ExecuteFunc>("execute");

        std::cout << "插件名: " << name_fn() << "\n";
        std::cout << "版本: " << version_fn() << "\n";
        std::cout << "计算 3 + 5 = " << exec_fn(3, 5) << "\n";

    } catch (const std::exception& e) {
        std::cerr << e.what() << "\n";
        return 1;
    }

    return 0;
}
```

## 3. 工厂模式插件系统

### 3.1 抽象接口（纯虚基类）

```cpp
// iplugin.h
#pragma once
#include <string>

class IPlugin {
public:
    virtual ~IPlugin() = default;
    virtual const char* name() const = 0;
    virtual int execute(int a, int b) = 0;
};
```

### 3.2 插件侧：导出工厂函数

```cpp
// add_plugin.cpp
#include "iplugin.h"

class AddPlugin : public IPlugin {
public:
    const char* name() const override { return "加法"; }
    int execute(int a, int b) override { return a + b; }
};

// C接口：创建/销毁
extern "C" {

#ifdef _WIN32
    __declspec(dllexport)
#else
    __attribute__((visibility("default")))
#endif
IPlugin* create_plugin() {
    return new AddPlugin();
}

#ifdef _WIN32
    __declspec(dllexport)
#else
    __attribute__((visibility("default")))
#endif
void destroy_plugin(IPlugin* p) {
    delete p;
}

}
```

### 3.3 主程序：插件管理器

```cpp
// plugin_manager.h
#pragma once
#include "iplugin.h"
#include "DynamicLib.h"     // 之前的封装
#include <memory>
#include <vector>
#include <string>
#include <filesystem>

namespace fs = std::filesystem;

class PluginManager {
public:
    void load_plugins(const std::string& directory) {
        for (const auto& entry : fs::directory_iterator(directory)) {
            if (!entry.is_regular_file()) continue;

            auto ext = entry.path().extension().string();
#ifdef _WIN32
            if (ext != ".dll") continue;
#else
            if (ext != ".so") continue;
#endif

            try {
                auto lib = std::make_unique<DynamicLib>(entry.path().string());

                using CreateFunc = IPlugin* (*)();
                using DestroyFunc = void (*)(IPlugin*);

                auto create  = lib->get_function<CreateFunc>("create_plugin");
                auto destroy = lib->get_function<DestroyFunc>("destroy_plugin");

                if (!create || !destroy) continue;

                IPlugin* raw_plugin = create();
                if (!raw_plugin) continue;

                // 用 shared_ptr + 自定义删除器管理
                auto plugin = std::shared_ptr<IPlugin>(raw_plugin, [destroy](IPlugin* p) {
                    destroy(p);
                });

                plugins_.push_back(plugin);
                libs_.push_back(std::move(lib));

                std::cout << "已加载插件: " << plugin->name() << "\n";

            } catch (...) {
                // 加载失败，跳过
            }
        }
    }

    IPlugin* find(const std::string& name) {
        for (auto& p : plugins_) {
            if (p->name() == name) return p.get();
        }
        return nullptr;
    }

    const auto& plugins() const { return plugins_; }

private:
    std::vector<std::unique_ptr<DynamicLib>> libs_;        // 保活动态库
    std::vector<std::shared_ptr<IPlugin>> plugins_;         // 插件实例
};
```

### 3.4 使用插件管理器

```cpp
int main() {
    PluginManager manager;
    manager.load_plugins("./plugins");  // 扫描plugins目录下所有.dll/.so

    // 列出所有插件
    for (const auto& p : manager.plugins()) {
        std::cout << p->name() << ": " << p->execute(10, 20) << "\n";
    }

    // 按名称查找使用
    if (auto* add = manager.find("加法")) {
        std::cout << "100 + 200 = " << add->execute(100, 200) << "\n";
    }

    return 0;
}
```

## 4. 常见陷阱 & 最佳实践

| 陷阱 | 说明 | 正确做法 |
|------|------|----------|
| **C++ ABI不兼容** | 不同编译器/版本的C++类布局不同 | 导出C接口（`extern "C"`），通过虚基类指针交互 |
| **跨模块内存管理** | 在libc中new、在libc++中delete→崩溃 | 提供配套的`destroy_plugin()`，在插件侧释放 |
| **不保活动态库** | 提前卸载导致函数指针和对象悬空 | 用 `unique_ptr<DynamicLib>` 保活，卸载顺序在后 |
| **符号名修饰** | C++ mangling导致dlsym找不到函数 | 导出函数用`extern "C"`包裹 |
| **Linux符号隐藏** | 默认全部导出，污染命名空间 | CMake设`CXX_VISIBILITY_PRESET hidden`，显式导出 |
| **STL跨模块传递** | 不同模块的std::string内存布局可能不同 | C接口用 `const char*`，或确保所有模块用同一编译器和STL |

**最佳实践：**

- 导出接口一律用 `extern "C"` 包裹
- 工厂函数返回纯虚基类指针，搭配 `destroy` 函数
- 动态库加载后用 `unique_ptr` 保活，确保在插件对象之后再析构
- CMake 设置 `CXX_VISIBILITY_PRESET hidden` 控制符号可见性
- 插件目录约定：`plugins/` 下放所有 `.dll/.so`
