# 09 — Qt 核心编程指南

## 1. 核心概念

Qt 是跨平台C++ GUI框架，但远不止GUI——它提供字符串、容器、网络、数据库、多线程、XML/JSON等全栈解决方案。

核心机制：
- **信号与槽**（Signals & Slots）：对象间通信的解耦机制，Qt的灵魂
- **事件循环**（QApplication::exec()）：驱动整个GUI程序
- **元对象系统**（MOC）：编译期代码生成，支撑信号槽/属性/反射
- **父子对象模型**：父对象析构时自动删除子对象，简化内存管理

**为什么用Qt？**一套代码跨 Windows/macOS/Linux，组件丰富，文档完善，工业级稳定性。

## 2. 快速上手

### CMake 引入

```cmake

```

## 3. 核心机制速查

### 3.1 信号与槽

```cpp
#include <QApplication>
#include <QPushButton>
#include <QLabel>
#include <QVBoxLayout>
#include <QWidget>

class MyWidget : public QWidget {
    Q_OBJECT  // 必须有，启用元对象系统

public:
    MyWidget(QWidget* parent = nullptr) : QWidget(parent) {
        auto* layout = new QVBoxLayout(this);

        label_ = new QLabel("计数: 0", this);
        auto* btn = new QPushButton("增加", this);

        layout->addWidget(label_);
        layout->addWidget(btn);

        // 连接信号到槽（5种写法）
        // 写法1：成员函数指针（推荐，编译期类型检查）
        connect(btn, &QPushButton::clicked, this, &MyWidget::onButtonClicked);

        // 写法2：Lambda（最灵活）
        connect(btn, &QPushButton::clicked, this, [this] {
            label_->setText(QString("点击了 %1 次").arg(++count_));
        });

        // 写法3：信号连信号
        // connect(slider, &QSlider::valueChanged, spinbox, &QSpinBox::setValue);

        // 写法4：旧式字符串（不推荐，运行时检查）
        // connect(btn, SIGNAL(clicked()), this, SLOT(onButtonClicked()));

        // 写法5：C++14泛型Lambda
        // connect(btn, &QPushButton::clicked, [](auto checked) { ... });
    }

private slots:         // slots 关键字 = public in Qt5+
    void onButtonClicked() {
        label_->setText(QString("计数: %1").arg(++count_));
    }

private:
    QLabel* label_;
    int count_ = 0;
};

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);
    MyWidget w;
    w.show();
    return app.exec();
}
```

```cpp
signals:
    void finished();
};

worker->moveToThread(thread);
```

### 3.5 自定义信号

```cpp
class DataFetcher : public QObject {
    Q_OBJECT
public:
    void fetch() {
        // 模拟异步获取数据
        emit dataReady(QStringList{"item1", "item2"});
        emit progressChanged(100);
    }

signals:
    void dataReady(const QStringList& data);
    void progressChanged(int percent);
    void errorOccurred(const QString& message);
```

## 4. 常见陷阱 & 最佳实践

| 陷阱 | 说明 | 正确做法 |
|------|------|----------|
| **忘记 Q_OBJECT** | 没有MOC处理，信号槽不工作，编译不报错 | 凡是继承QObject且用信号槽的类必须加 |
| **跨线程操作GUI** | 子线程直接操作widget → 崩溃或未定义 | 用信号槽传递，或 `QMetaObject::invokeMethod` |
| **对象树管理误区** | 栈对象设置了parent会导致双重析构 | 栈对象不设parent，堆对象设parent自动管理 |
| **connect 生命周期** | 连接到已删除对象的槽 | 用context参数自动断开：`connect(sender, sig, receiver, slot)` |
| **事件循环阻塞** | 槽中做耗时操作导致UI卡死 | 耗时任务放QThread或用 `QCoreApplication::processEvents()` |
| **MOC解析旧宏** | SIGNAL/SLOT宏错误不编译报错 | Qt5+ 用函数指针写法，编译期检查 |
| **CMAKE_AUTOMOC未开启** | Q_OBJECT类编译链接错误 | CMakeLists.txt中设置 `set(CMAKE_AUTOMOC ON)` |

**最佳实践：**

- connect优先用函数指针写法，Lambda次之，避免字符串写法
- 窗口/控件用 `new` + parent 管理生命周期，避免手动 delete
- 信号命名用过去式：`dataReady()` / 槽命名用动词：`onDataReady()`
- 耗时操作放 `QThread`，通过信号槽回传结果到主线程更新UI

## 5. 生产项目典型用法

### 场景：简单 Todo 应用窗口

```cpp
#include <QApplication>
#include <QMainWindow>
#include <QListWidget>
#include <QLineEdit>
#include <QPushButton>
#include <QVBoxLayout>
#include <QHBoxLayout>
#include <QWidget>
#include <QMenuBar>
#include <QMessageBox>
#include <QFile>
#include <QJsonDocument>
#include <QJsonArray>

class TodoWindow : public QMainWindow {
    Q_OBJECT
public:
    TodoWindow(QWidget* parent = nullptr) : QMainWindow(parent) {
        setWindowTitle("Todo List");
        resize(400, 500);

        auto* central = new QWidget(this);
        auto* layout = new QVBoxLayout(central);

        // 输入区
        auto* input_layout = new QHBoxLayout;
        input_ = new QLineEdit;
        input_->setPlaceholderText("输入新任务...");
        auto* add_btn = new QPushButton("添加");
        input_layout->addWidget(input_);
        input_layout->addWidget(add_btn);

        // 列表
        list_ = new QListWidget;

        // 按钮区
        auto* btn_layout = new QHBoxLayout;
        auto* del_btn = new QPushButton("删除选中");
        auto* clear_btn = new QPushButton("清空全部");
        btn_layout->addWidget(del_btn);
        btn_layout->addWidget(clear_btn);

        layout->addLayout(input_layout);
        layout->addWidget(list_);
        layout->addLayout(btn_layout);
        setCentralWidget(central);

        // 连接信号
        connect(add_btn, &QPushButton::clicked, this, &TodoWindow::addItem);
        connect(input_, &QLineEdit::returnPressed, this, &TodoWindow::addItem);
        connect(del_btn, &QPushButton::clicked, this, [this] {
            delete list_->takeItem(list_->currentRow());
        });
        connect(clear_btn, &QPushButton::clicked, list_, &QListWidget::clear);

        // 菜单
        auto* file_menu = menuBar()->addMenu("文件");
        file_menu->addAction("保存", this, &TodoWindow::saveToFile);
        file_menu->addAction("加载", this, &TodoWindow::loadFromFile);
    }

private slots:
    void addItem() {
        QString text = input_->text().trimmed();
        if (!text.isEmpty()) {
            list_->addItem(text);
            input_->clear();
        }
    }

    void saveToFile() {
        QJsonArray arr;
        for (int i = 0; i < list_->count(); ++i)
            arr.append(list_->item(i)->text());

        QFile file("todos.json");
        if (file.open(QIODevice::WriteOnly)) {
            file.write(QJsonDocument(arr).toJson());
            file.close();
            QMessageBox::information(this, "提示", "保存成功");
        }
    }

    void loadFromFile() {
        QFile file("todos.json");
        if (file.open(QIODevice::ReadOnly)) {
            auto arr = QJsonDocument::fromJson(file.readAll()).array();
            list_->clear();
            for (const auto& val : arr)
                list_->addItem(val.toString());
            file.close();
        }
    }

private:
    QLineEdit* input_;
    QListWidget* list_;
};

int main(int argc, char* argv[]) {
    QApplication app(argc, argv);
    TodoWindow w;
    w.show();
    return app.exec();
}

#include "main.moc"  // 如果CMAKE_AUTOMOC OFF需要手动include
```
